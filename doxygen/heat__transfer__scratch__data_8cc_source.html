9template <int dim>
10void
11HeatTransferScratchData<dim>::allocate()
12{
13 // Initialize size of arrays
14 this->n_q_points = fe_values_T.get_quadrature().size();
15 this->n_dofs = fe_values_T.get_fe().n_dofs_per_cell();
16
17 // Initialize arrays related to quadrature
18 this->JxW.reinit(n_q_points);
19
20 // Forcing term array
21 this->source = std::vector<double>(n_q_points);
22
23 // Initialize arrays related to velocity and pressure
24 // Velocity
25 this->velocity_values = std::vector<Tensor<1, dim>>(n_q_points);
26 this->velocity_gradient_values = std::vector<Tensor<2, dim>>(n_q_points);
27 this->shear_rate_values = std::vector<double>(n_q_points);
28 // Pressure
29 this->pressure_values = std::vector<double>(n_q_points);
30
31 // Initialize arrays related to temperature
32 this->present_temperature_values = std::vector<double>(n_q_points);
33 this->temperature_gradients = std::vector<Tensor<1, dim>>(n_q_points);
34 this->present_temperature_laplacians = std::vector<double>(n_q_points);
35
36 // Initial arrays for physical properties
37 this->specific_heat = std::vector<double>(n_q_points);
38 this->grad_specific_heat_temperature = std::vector<double>(n_q_points);
39 this->thermal_conductivity = std::vector<double>(n_q_points);
40 this->density = std::vector<double>(n_q_points);
41 this->dynamic_viscosity = std::vector<double>(n_q_points);
42
43 // Velocity for BDF schemes
44 this->previous_temperature_values =
45 std::vector<std::vector<double>>(maximum_number_of_previous_solutions(),
46 std::vector<double>(n_q_points));
47 this->previous_temperature_gradients =
48 std::vector<std::vector<Tensor<1, dim>>>(
49 maximum_number_of_previous_solutions(),
50 std::vector<Tensor<1, dim>>(n_q_points));
51
52 // Initialize arrays related to shape functions
53 // Velocity shape functions
54 this->phi_T.reinit(TableIndices<2>(n_q_points, n_dofs));
55 this->grad_phi_T.reinit(TableIndices<2>(n_q_points, n_dofs));
56 this->hess_phi_T.reinit(TableIndices<2>(n_q_points, n_dofs));
57 this->laplacian_phi_T.reinit(TableIndices<2>(n_q_points, n_dofs));
58
59 // Physical properties
60 fields.insert(
61 std::pair<field, std::vector<double>>(field::temperature, n_q_points));
62 fields.insert(
63 std::pair<field, std::vector<double>>(field::temperature_p1, n_q_points));
64 fields.insert(
65 std::pair<field, std::vector<double>>(field::temperature_p2, n_q_points));
66 fields.insert(
67 std::pair<field, std::vector<double>>(field::shear_rate, n_q_points));
68 fields.insert(
69 std::pair<field, std::vector<double>>(field::pressure, n_q_points));
70}

74HeatTransferScratchData<dim>::enable_cls(

83
84 // Allocate CLS values

86 filtered_phase_gradient_values =
87 std::vector<Tensor<1, dim>>(this->n_q_points);
88
89 phase_gradient_values = std::vector<Tensor<1, dim>>(this->n_q_points);
90
91 // Allocate physical properties
92 specific_heat_0 = std::vector<double>(n_q_points);
93 density_0 = std::vector<double>(n_q_points);
94 thermal_conductivity_0 = std::vector<double>(n_q_points);
95 dynamic_viscosity_0 = std::vector<double>(n_q_points);
96 grad_specific_heat_temperature_0 = std::vector<double>(n_q_points);
97
98 specific_heat_1 = std::vector<double>(n_q_points);
99 density_1 = std::vector<double>(n_q_points);
100 thermal_conductivity_1 = std::vector<double>(n_q_points);
101 dynamic_viscosity_1 = std::vector<double>(n_q_points);
102 grad_specific_heat_temperature_1 = std::vector<double>(n_q_points);
103
104 // Create filter
105 filter = ConservativeLevelSetFilterBase::model_cast(phase_filter_parameters);
106}

145void
146HeatTransferScratchData<dim>::enable_time_harmonic_maxwell(
147 const FiniteElement<dim> &fe,
148 const Quadrature<dim> &quadrature,
149 const Mapping<dim> &mapping)
150{
151 gather_time_harmonic_maxwell = true;
152 fe_values_time_harmonic_maxwell =
153 std::make_shared<FEValues<dim>>(mapping, fe, quadrature, update_values);
154
155 // Allocate time-harmonic Maxwell values
156 this->electric_field_real_values =
157 std::vector<Tensor<1, dim>>(this->n_q_points);
158 this->electric_field_imag_values =
159 std::vector<Tensor<1, dim>>(this->n_q_points);
160 this->magnetic_field_real_values =
161 std::vector<Tensor<1, dim>>(this->n_q_points);
162 this->magnetic_field_imag_values =
163 std::vector<Tensor<1, dim>>(this->n_q_points);
164
165 // Allocate physical properties
166 this->electric_conductivity = std::vector<double>(this->n_q_points);
167 this->magnetic_permeability_imag = std::vector<double>(this->n_q_points);
168 this->electric_permittivity_imag = std::vector<double>(this->n_q_points);
169
170 // Define extractors for the time-harmonic Maxwell fields
171 this->extractor_E_real = FEValuesExtractors::Vector(0);
172 this->extractor_E_imag = FEValuesExtractors::Vector(dim);
173 this->extractor_H_real = FEValuesExtractors::Vector(2 * dim);
174 this->extractor_H_imag = FEValuesExtractors::Vector(3 * dim);
175}

177template <int dim>
178void
179HeatTransferScratchData<dim>::calculate_physical_properties()
180{
181 set_field_vector(field::temperature,
182 this->present_temperature_values,
183 this->fields);
184
185 if (properties_manager.field_is_required(field::temperature_p1))
186 set_field_vector(field::temperature_p1,
187 this->previous_temperature_values[0],
188 this->fields);
189
190 if (properties_manager.field_is_required(field::temperature_p2))
191 set_field_vector(field::temperature_p2,
192 this->previous_temperature_values[1],
193 this->fields);
194
195 if (properties_manager.field_is_required(field::shear_rate))
196 {
197 for (unsigned int q = 0; q < n_q_points; ++q)
198 {
199 // Calculate shear rate (at each q)
200 const Tensor<2, dim> shear_rate_tensor =
201 velocity_gradient_values[q] +
202 transpose(velocity_gradient_values[q]);
203
204 // Calculate the shear rate magnitude
205 shear_rate_values[q] =
206 calculate_shear_rate_magnitude(shear_rate_tensor);
207 }
208
209 set_field_vector(field::shear_rate, shear_rate_values, this->fields);
210 }
211
212 if (properties_manager.field_is_required(field::pressure))
213 {
214 set_field_vector(field::pressure, this->pressure_values, this->fields);
215 }
216
217 if (material_id < 1 || properties_manager.get_number_of_solids() < 1)
218 {
219 switch (properties_manager.get_number_of_fluids())
220 {
221 case 1:
222 {
223 const auto density_model = properties_manager.get_density();
224 const auto specific_heat_model =
225 properties_manager.get_specific_heat();
226 const auto thermal_conductivity_model =
227 properties_manager.get_thermal_conductivity();
228 const auto rheology_model = properties_manager.get_rheology();
229
230 density_model->vector_value(fields, density);
231 specific_heat_model->vector_value(fields, specific_heat);
232 specific_heat_model->vector_jacobian(
233 fields, field::temperature, grad_specific_heat_temperature);
234 thermal_conductivity_model->vector_value(fields,
235 thermal_conductivity);
236 rheology_model->get_dynamic_viscosity_vector(
237 density_model->get_density_ref(), fields, dynamic_viscosity);
238
239 if (gather_time_harmonic_maxwell)
240 {
241 const auto electric_conductivity_model =
242 properties_manager.get_electric_conductivity();
243 const auto magnetic_permeability_model =
244 properties_manager.get_magnetic_permeability_imag();
245 const auto electric_permittivity_model =
246 properties_manager.get_electric_permittivity_imag();
247
248 electric_conductivity_model->vector_value(
249 fields, electric_conductivity);
250 magnetic_permeability_model->vector_value(
251 fields, magnetic_permeability_imag);
252 electric_permittivity_model->vector_value(
253 fields, electric_permittivity_imag);
254 }
255
256 break;
257 }
258 case 2:
259 {
260 const auto density_models =
261 properties_manager.get_density_vector();
262 const auto specific_heat_models =
263 properties_manager.get_specific_heat_vector();
264 const auto thermal_conductivity_models =
265 properties_manager.get_thermal_conductivity_vector();
266 const auto rheology_models =
267 properties_manager.get_rheology_vector();
268
269 density_models[0]->vector_value(fields, density_0);
270 specific_heat_models[0]->vector_value(fields, specific_heat_0);
271 thermal_conductivity_models[0]->vector_value(
272 fields, thermal_conductivity_0);
273 rheology_models[0]->get_dynamic_viscosity_vector(
274 density_models[0]->get_density_ref(),
275 fields,
276 dynamic_viscosity_0);
277 specific_heat_models[0]->vector_jacobian(
278 fields, field::temperature, grad_specific_heat_temperature_0);
279
280 density_models[1]->vector_value(fields, density_1);
281 specific_heat_models[1]->vector_value(fields, specific_heat_1);
282 thermal_conductivity_models[1]->vector_value(
283 fields, thermal_conductivity_1);
284 rheology_models[1]->get_dynamic_viscosity_vector(
285 density_models[1]->get_density_ref(),
286 fields,
287 dynamic_viscosity_1);
288 specific_heat_models[1]->vector_jacobian(
289 fields, field::temperature, grad_specific_heat_temperature_1);
290
291 // Blend the physical properties using the CLS field
292 for (unsigned int q = 0; q < this->n_q_points; ++q)
293 {
294 auto filtered_phase_value = this->filtered_phase_values[q];
295
296 density[q] = calculate_point_property(filtered_phase_value,
297 this->density_0[q],
298 this->density_1[q]);
299
300 specific_heat[q] =
301 calculate_point_property(filtered_phase_value,
302 this->specific_heat_0[q],
303 this->specific_heat_1[q]);
304
305 thermal_conductivity[q] =
306 calculate_point_property(filtered_phase_value,
307 this->thermal_conductivity_0[q],
308 this->thermal_conductivity_1[q]);
309
310 dynamic_viscosity[q] =
311 calculate_point_property(filtered_phase_value,
312 this->dynamic_viscosity_0[q],
313 this->dynamic_viscosity_1[q]);
314
315 grad_specific_heat_temperature[q] = calculate_point_property(
316 filtered_phase_value,
317 this->grad_specific_heat_temperature_0[q],
318 this->grad_specific_heat_temperature_1[q]);
319 }
320 }
321 break;
322 default:
323 throw std::runtime_error("Unsupported number of fluids (>2)");
324 }
325 }
326 else
327 {
328 const auto density_model = properties_manager.get_density(0, material_id);
329 const auto specific_heat_model =
330 properties_manager.get_specific_heat(0, material_id);
331 const auto thermal_conductivity_model =
332 properties_manager.get_thermal_conductivity(0, material_id);
333 const auto rheology_model =
334 properties_manager.get_rheology(0, material_id);
335
336 density_model->vector_value(fields, density);
337 specific_heat_model->vector_value(fields, specific_heat);
338 specific_heat_model->vector_jacobian(fields,
339 field::temperature,
340 grad_specific_heat_temperature);
341 thermal_conductivity_model->vector_value(fields, thermal_conductivity);
342 rheology_model->get_dynamic_viscosity_vector(
343 density_model->get_density_ref(), fields, dynamic_viscosity);
344
345 if (gather_time_harmonic_maxwell)
346 {
347 const auto electric_conductivity_model =
348 properties_manager.get_electric_conductivity(0, material_id);
349 const auto magnetic_permeability_model =
350 properties_manager.get_magnetic_permeability_imag(0, material_id);
351 const auto electric_permittivity_model =
352 properties_manager.get_electric_permittivity_imag(0, material_id);
353
354 electric_conductivity_model->vector_value(fields,
355 electric_conductivity);
356 magnetic_permeability_model->vector_value(fields,
357 magnetic_permeability_imag);
358 electric_permittivity_model->vector_value(fields,
359 electric_permittivity_imag);
360 }
361 }
362}
