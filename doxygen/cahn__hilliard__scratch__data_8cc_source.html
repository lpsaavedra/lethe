7template <int dim>
8void
9CahnHilliardScratchData<dim>::allocate()
10{
11 // Initialize size of arrays
12 this->n_q_points = fe_values_cahn_hilliard.get_quadrature().size();
13 this->n_dofs = fe_values_cahn_hilliard.get_fe().n_dofs_per_cell();
14
15 // Initialize arrays related to quadrature
16 this->JxW.reinit(n_q_points);
17
18 // Forcing term array
19 this->source_phase_order = std::vector<double>(n_q_points);
20 this->source_chemical_potential = std::vector<double>(n_q_points);
21
22 // Initialize arrays related to phase order and chemical potential
23 this->phase_order.component = 0;
24 this->chemical_potential.component = 1;
25
26 // Variables for the Cahn-Hilliard equations
27 this->phase_order_values = std::vector<double>(n_q_points);
28 this->phase_order_gradients = std::vector<Tensor<1, dim>>(n_q_points);
29 this->phase_order_laplacians = std::vector<double>(n_q_points);
30 this->chemical_potential_values = std::vector<double>(n_q_points);
31 this->chemical_potential_gradients = std::vector<Tensor<1, dim>>(n_q_points);
32 this->chemical_potential_laplacians = std::vector<double>(n_q_points);
33
34 // Phase order for BDF schemes
35 this->previous_phase_order_values =
36 std::vector<std::vector<double>>(maximum_number_of_previous_solutions(),
37 std::vector<double>(n_q_points));
38
39 this->previous_chemical_potential_values =
40 std::vector<std::vector<double>>(maximum_number_of_previous_solutions(),
41 std::vector<double>(n_q_points));
42
43 // Initialize arrays related to shape functions
44 // Phase-order shape functions
45 this->phi_phase.reinit(n_q_points, n_dofs);
46 this->grad_phi_phase.reinit(n_q_points, n_dofs);
47 this->hess_phi_phase.reinit(n_q_points, n_dofs);
48 this->laplacian_phi_phase.reinit(n_q_points, n_dofs);
49
50 // Chemical potential shape functions
51 this->phi_potential.reinit(n_q_points, n_dofs);
52 this->grad_phi_potential.reinit(n_q_points, n_dofs);
53 this->hess_phi_potential.reinit(n_q_points, n_dofs);
54 this->laplacian_phi_potential.reinit(n_q_points, n_dofs);
55
56 // Velocity values
57 this->velocity_values = std::vector<Tensor<1, dim>>(this->n_q_points);
58 this->previous_velocity_values = std::vector<std::vector<Tensor<1, dim>>>(
59 maximum_number_of_previous_solutions(),
60 std::vector<Tensor<1, dim>>(this->n_q_points));
61 this->velocity_gradient_values =
62 std::vector<Tensor<2, dim>>(this->n_q_points);
63
64 // Allocate physical properties
65 this->surface_tension = std::vector<double>(n_q_points);
66 this->mobility_cahn_hilliard = std::vector<double>(n_q_points);
67 this->mobility_cahn_hilliard_gradient = std::vector<double>(n_q_points);
68
69 // Physical properties
70 fields.insert(
71 std::pair<field, std::vector<double>>(field::phase_order_cahn_hilliard,
72 n_q_points));
73 fields.insert(std::pair<field, std::vector<double>>(
74 field::phase_order_cahn_hilliard_filtered, n_q_points));
75}

77template <int dim>
78void
79CahnHilliardScratchData<dim>::calculate_physical_properties()
80{
81 switch (properties_manager.get_number_of_fluids())
82 {
83 case 1:
84 {
85 throw std::runtime_error("Unsupported number of fluids (<2)");
86 }
87 case 2:
88 {
89 set_field_vector(field::phase_order_cahn_hilliard,
90 this->phase_order_values,
91 this->fields);
92
93 // Gather properties from material interactions
94 const auto material_interaction_id =
95 properties_manager.get_material_interaction_id(
96 material_interactions_type::fluid_fluid, 0, 1);
97
98 // Gather surface tension
99 const auto surface_tension_model =
100 properties_manager.get_surface_tension(material_interaction_id);
101 surface_tension_model->vector_value(fields, surface_tension);
102
103 // Gather mobility
104 const auto mobility_cahn_hilliard_model =
105 properties_manager.get_mobility_cahn_hilliard(
106 material_interaction_id);
107 mobility_cahn_hilliard_model->vector_value(fields,
108 mobility_cahn_hilliard);
109 mobility_cahn_hilliard_model->vector_jacobian(
110 fields, phase_order_cahn_hilliard, mobility_cahn_hilliard_gradient);
111
112 break;
113 }
114 default:
115 throw std::runtime_error("Unsupported number of fluids (>2)");
116 }
117}
