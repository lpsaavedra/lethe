9template <int dim>
10void
11NavierStokesScratchData<dim>::allocate()
12{
13 // Initialize size of arrays
14 this->n_q_points = fe_values.get_quadrature().size();
15 this->n_dofs = fe_values.get_fe().n_dofs_per_cell();
16
17 // Initialize arrays related to quadrature
18 this->JxW.reinit(n_q_points);
19
20 // Initialize component array
21 this->components = std::vector<unsigned int>(n_dofs);
22
23 // Forcing term array
24 this->rhs_force =
25 std::vector<Vector<double>>(n_q_points, Vector<double>(dim + 1));
26 this->force = std::vector<Tensor<1, dim>>(n_q_points);
27 this->mass_source = std::vector<double>(n_q_points);
28
29 // Initialize arrays related to velocity and pressure
30 this->velocities.first_vector_component = 0;
31 this->pressure.component = dim;
32
33 // Velocity
34 this->velocity_values = std::vector<Tensor<1, dim>>(n_q_points);
35 this->velocity_divergences = std::vector<double>(n_q_points);
36 this->velocity_gradients = std::vector<Tensor<2, dim>>(n_q_points);
37 this->velocity_laplacians = std::vector<Tensor<1, dim>>(n_q_points);
38 this->velocity_hessians = std::vector<Tensor<3, dim>>(n_q_points);
39 this->velocity_gradient_divergence = std::vector<Tensor<1, dim>>(n_q_points);
40 this->shear_rate = std::vector<double>(n_q_points);
41 this->advective_velocity = std::vector<Tensor<1, dim>>(n_q_points);
42
43 // For SDIRK method: sum(a_ij * k_j)
44 if (this->simulation_control->is_sdirk())
45 {
46 this->sdirk_stage_sum = std::vector<Tensor<1, dim>>(n_q_points);
47 }
48
49 // Velocity for BDF schemes
50 this->previous_velocity_values = std::vector<std::vector<Tensor<1, dim>>>(
51 maximum_number_of_previous_solutions(),
52 std::vector<Tensor<1, dim>>(n_q_points));
53
54 // Pressure
55 this->pressure_values = std::vector<double>(n_q_points);
56 this->pressure_gradients = std::vector<Tensor<1, dim>>(n_q_points);
57 this->pressure_scaling_factor = 1;
58
59 // Pressure for BDF schemes (compressible Navier-Stokes)
60 this->previous_pressure_values =
61 std::vector<std::vector<double>>(maximum_number_of_previous_solutions(),
62 std::vector<double>(n_q_points));
63 // Initialize arrays related to shape functions
64 // Velocity shape functions
65 this->phi_u.reinit(n_q_points, n_dofs);
66 this->grad_phi_u.reinit(n_q_points, n_dofs);
67 this->div_phi_u.reinit(n_q_points, n_dofs);
68 this->hess_phi_u.reinit(n_q_points, n_dofs);
69 this->laplacian_phi_u.reinit(n_q_points, n_dofs);
70 this->gradient_divergence_phi_u.reinit(n_q_points, n_dofs);
71
72 // Pressure shape functions
73 this->phi_p.reinit(n_q_points, n_dofs);
74 this->grad_phi_p.reinit(n_q_points, n_dofs);
75
76 // Physical properties
77 fields.insert(
78 std::pair<field, std::vector<double>>(field::pressure, n_q_points));
79 fields.insert(
80 std::pair<field, std::vector<double>>(field::shear_rate, n_q_points));
81
82 density = std::vector<double>(n_q_points);
83 dynamic_viscosity = std::vector<double>(n_q_points);
84 kinematic_viscosity = std::vector<double>(n_q_points);
85 dynamic_viscosity_for_stabilization = std::vector<double>(n_q_points);
86 kinematic_viscosity_for_stabilization = std::vector<double>(n_q_points);
87 thermal_expansion = std::vector<double>(n_q_points);
88 density_thermal_expansion = std::vector<double>(n_q_points);
89 grad_kinematic_viscosity_shear_rate = std::vector<double>(n_q_points);
90
91 previous_density =
92 std::vector<std::vector<double>>(maximum_number_of_previous_solutions(),
93 std::vector<double>(this->n_q_points));
94}

415template <int dim>
416void
417NavierStokesScratchData<dim>::reinit_ale(const Parameters::ALE<dim> &ale)
418{
419 Tensor<1, dim> velocity_ale;
420 std::shared_ptr<Functions::ParsedFunction<dim>> velocity_ale_function =
421 ale.velocity;
422 Vector<double> velocity_ale_vector(dim);
423
424 for (unsigned int q = 0; q < n_q_points; ++q)
425 {
426 velocity_ale_function->vector_value(quadrature_points[q],
427 velocity_ale_vector);
428 for (int d = 0; d < dim; ++d)
429 velocity_ale[d] = velocity_ale_vector[d];
430
431 // Subtract u_ale from the advective velocity
432 this->advective_velocity[q] -= velocity_ale;
433 }
434}

443template <int dim>
444void
445NavierStokesScratchData<dim>::reinit_mortar(
446 const typename DoFHandler<dim>::active_cell_iterator &cell,
447 const Parameters::Mortar<dim> &mortar_parameters)
448{
449 // Get updated rotor angular velocity
450 mortar_parameters.rotor_angular_velocity->set_time(
451 this->simulation_control->get_current_time());
452 const double rotor_angular_velocity =
453 mortar_parameters.rotor_angular_velocity->value(Point<dim>());
454
455 // Apply prescribed rotor angular velocity only at rotor cells
456 // (material_id = 1)
457 double omega = 0.0;
458 if (cell->material_id() == 1)
459 omega = rotor_angular_velocity;
460
461 // The mortar implementation is not defined for more than two
462 // materials. Throw when running debug mode as a security measure to prevent
463 // this case
464 Assert(
465 cell->material_id() < 2,
466 ExcMessage(
467 "The material id in a cell was identified as equal or greater than 2, however the mortar implementation is not defined for more than two materials."));
468
469 // Compute rotor linear velocity at quadrature points
470 rotor_linear_velocity_values = std::vector<Tensor<1, dim>>(this->n_q_points);
471 for (unsigned int q = 0; q < n_q_points; ++q)
472 {
473 // Shift quadrature point by the center of rotation
474 const auto p =
475 fe_values.quadrature_point(q) - mortar_parameters.center_of_rotation;
476
477 // Compute terms of u_ale
478 rotor_linear_velocity_values[q] =
479 LetheGridTools::angular_to_linear_velocity(
480 omega, p, mortar_parameters.rotation_axis_direction);
481
482 // Subtract u_ale from the advective velocity
483 this->advective_velocity[q] -= rotor_linear_velocity_values[q];
484 }
485}

512template <int dim>
513void
514NavierStokesScratchData<dim>::calculate_physical_properties()
515{
516 if (properties_manager.field_is_required(field::temperature) &&
517 gather_temperature)
518 {
519 set_field_vector(field::temperature,
520 this->temperature_values,
521 this->fields);
522 }
523
524 if (properties_manager.field_is_required(field::pressure))
525 {
526 set_field_vector(field::pressure, this->pressure_values, this->fields);
527 }
528
529 if (properties_manager.field_is_required(field::shear_rate))
530 {
531 for (unsigned int q = 0; q < n_q_points; ++q)
532 {
533 // Calculate shear rate (at each q)
534 const Tensor<2, dim> shear_rate_tensor =
535 velocity_gradients[q] + transpose(velocity_gradients[q]);
536
537 // Calculate the shear rate magnitude
538 shear_rate[q] = calculate_shear_rate_magnitude(shear_rate_tensor);
539 }
540
541 set_field_vector(field::shear_rate, shear_rate, this->fields);
542 }
543
544 switch (properties_manager.get_number_of_fluids())
545 {
546 case 1:
547 {
548 // In the case of an incompressible flow, viscosity is the only
549 // required property
550 const auto rheology_model = properties_manager.get_rheology();
551 rheology_model->vector_value(fields, kinematic_viscosity);
552
553 rheology_model->get_kinematic_viscosity_for_stabilization_vector(
554 fields, kinematic_viscosity_for_stabilization);
555
556 kinematic_viscosity_scale =
557 rheology_model->get_kinematic_viscosity_scale();
558
559 if (!properties_manager.density_is_constant())
560 {
561 // For a weakly compressible flow, density variations will play a
562 // role
563 const auto density_model = properties_manager.get_density();
564 density_model->vector_value(fields, density);
565 density_psi = density_model->get_psi();
566 density_ref = density_model->get_density_ref();
567 // Dynamic viscosity is also necessary for compressible flows
568 rheology_model->get_dynamic_viscosity_vector(density_ref,
569 fields,
570 dynamic_viscosity);
571 rheology_model->get_dynamic_viscosity_for_stabilization_vector(
572 density_ref, fields, dynamic_viscosity_for_stabilization);
573 }
574 else
575 {
576 density_scale = properties_manager.get_density_scale();
577 }
578
579 if (properties_manager.is_non_newtonian())
580 {
581 // Calculate derivative of viscosity with respect to shear rate
582 rheology_model->vector_jacobian(
583 fields, field::shear_rate, grad_kinematic_viscosity_shear_rate);
584 }
585
586 if (gather_temperature)
587 {
588 const auto thermal_expansion_model =
589 properties_manager.get_thermal_expansion();
590 thermal_expansion_model->vector_value(fields, thermal_expansion);
591 }
592 break;
593 }
594 case 2:
595 {
596 // We need both density and viscosity
597 const auto density_model_0 = properties_manager.get_density(0);
598 const auto rheology_model_0 = properties_manager.get_rheology(0);
599 const auto density_model_1 = properties_manager.get_density(1);
600 const auto rheology_model_1 = properties_manager.get_rheology(1);
601
602 density_ref_0 = density_model_0->get_density_ref();
603 density_ref_1 = density_model_1->get_density_ref();
604
605 kinematic_viscosity_scale_0 =
606 rheology_model_0->get_kinematic_viscosity_scale();
607 kinematic_viscosity_scale_1 =
608 rheology_model_1->get_kinematic_viscosity_scale();
609
610 kinematic_viscosity_scale =
611 std::max(kinematic_viscosity_scale_0, kinematic_viscosity_scale_1);
612
613 // Gather properties from material interactions if necessary
614 if (properties_manager.get_number_of_material_interactions() > 0)
615 {
616 const auto material_interaction_id =
617 properties_manager.get_material_interaction_id(
618 material_interactions_type::fluid_fluid, 0, 1);
619 // Gather surface tension
620 const auto surface_tension_model =
621 properties_manager.get_surface_tension(material_interaction_id);
622 surface_tension_model->vector_value(fields, surface_tension);
623 // Gather surface tension gradient only if necessary
624 if (!properties_manager.surface_tension_is_constant())
625 surface_tension_model->vector_jacobian(
626 fields, field::temperature, surface_tension_gradient);
627 }
628
629 density_model_0->vector_value(fields, density_0);
630 rheology_model_0->get_dynamic_viscosity_vector(density_ref_0,
631 fields,
632 dynamic_viscosity_0);
633 rheology_model_0->get_dynamic_viscosity_for_stabilization_vector(
634 density_ref_0, fields, dynamic_viscosity_for_stabilization_0);
635
636 density_model_1->vector_value(fields, density_1);
637 rheology_model_1->get_dynamic_viscosity_vector(density_ref_1,
638 fields,
639 dynamic_viscosity_1);
640 rheology_model_1->get_dynamic_viscosity_for_stabilization_vector(
641 density_ref_1, fields, dynamic_viscosity_for_stabilization_1);
642 if (gather_temperature)
643 {
644 const auto thermal_expansion_model_0 =
645 properties_manager.get_thermal_expansion(0);
646 const auto thermal_expansion_model_1 =
647 properties_manager.get_thermal_expansion(1);
648 thermal_expansion_model_0->vector_value(fields,
649 thermal_expansion_0);
650 thermal_expansion_model_1->vector_value(fields,
651 thermal_expansion_1);
652 }
653
654 if (gather_cls && !gather_cahn_hilliard)
655 {
656 for (unsigned int q = 0; q < this->n_q_points; ++q)
657 {
658 double filtered_phase_value = this->filtered_phase_values[q];
659
660 density[q] = calculate_point_property(filtered_phase_value,
661 this->density_0[q],
662 this->density_1[q]);
663
664 dynamic_viscosity[q] =
665 calculate_point_property(filtered_phase_value,
666 this->dynamic_viscosity_0[q],
667 this->dynamic_viscosity_1[q]);
668
669 kinematic_viscosity[q] = dynamic_viscosity[q] / density[q];
670
671 dynamic_viscosity_for_stabilization[q] =
672 calculate_point_property(
673 filtered_phase_value,
674 this->dynamic_viscosity_for_stabilization_0[q],
675 this->dynamic_viscosity_for_stabilization_1[q]);
676
677 thermal_expansion[q] =
678 calculate_point_property(filtered_phase_value,
679 this->thermal_expansion_0[q],
680 this->thermal_expansion_1[q]);
681
682 // Phase interpolate the product rho*beta per-phase to avoid
683 // cross term error from bilinear phase interpolation:
684 // interp_lin(rho)*interp_lin(beta) != interp_lin(rho*beta)
685 density_thermal_expansion[q] = calculate_point_property(
686 filtered_phase_value,
687 this->density_0[q] * this->thermal_expansion_0[q],
688 this->density_1[q] * this->thermal_expansion_1[q]);
689 }
690
691 // Gather density_psi for isothermal compressible NS equations
692 if (!properties_manager.density_is_constant())
693 {
694 density_psi_0 = density_model_0->get_psi();
695 density_psi_1 = density_model_1->get_psi();
696
697 for (unsigned int q = 0; q < this->n_q_points; ++q)
698 {
699 double filtered_phase_value =
700 this->filtered_phase_values[q];
701 // To avoid calculating twice in the assemblers
702 compressibility_multiplier[q] =
703 filtered_phase_value *
704 (density_0[q] / density_1[q] * density_psi_1 -
705 density_psi_0) +
706 density_psi_0;
707 }
708 }
709
710 for (unsigned p = 0; p < previous_phase_values.size(); ++p)
711 {
712 // Blend the physical properties using the CLS field
713 for (unsigned int q = 0; q < this->n_q_points; ++q)
714 {
715 // Calculate previous density (right now assumes constant
716 // density model per phase)
717 previous_density[p][q] = calculate_point_property(
718 filter->filter_phase(this->previous_phase_values[p][q]),
719 this->density_0[q],
720 this->density_1[q]);
721 }
722 }
723 break;
724 }
725 else if (gather_cahn_hilliard && !gather_cls)
726 {
727 // Blend the physical properties using the CahnHilliard field
728 for (unsigned int q = 0; q < this->n_q_points; ++q)
729 {
730 double phase_order_cahn_hilliard_value =
731 this->filtered_phase_order_cahn_hilliard_values[q];
732
733 density[q] = calculate_point_property_cahn_hilliard(
734 phase_order_cahn_hilliard_value,
735 this->density_0[q],
736 this->density_1[q]);
737
738 dynamic_viscosity[q] = calculate_point_property_cahn_hilliard(
739 phase_order_cahn_hilliard_value,
740 this->dynamic_viscosity_0[q],
741 this->dynamic_viscosity_1[q]);
742
743 dynamic_viscosity_for_stabilization[q] =
744 calculate_point_property_cahn_hilliard(
745 phase_order_cahn_hilliard_value,
746 this->dynamic_viscosity_for_stabilization_0[q],
747 this->dynamic_viscosity_for_stabilization_1[q]);
748 }
749 break;
750 }
751 break;
752 }
753 default:
754 throw std::runtime_error("Unsupported number of fluids (>2)");
755 }
